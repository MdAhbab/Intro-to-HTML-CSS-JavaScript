Conditionals

const score = 85;

if (score >= 90) {
  console.log('A');
} else if (score >= 80) {
  console.log('B');
} else {
  console.log('C or below');
}

const result = score >= 60 ? 'pass' : 'fail';
console.log(result);

Loops

// for
for (let i = 0; i < 3; i++) {
  console.log(i);
}

// while
let n = 3;
while (n > 0) {
  console.log(n);
  n--;
}

// for..of (arrays/iterables)
for (const tag of ['html','css','js']) {
  console.log(tag);
}

Truthy and Falsy

Falsy values: false, 0, '', null, undefined, NaN. Everything else is truthy.

const input = '';
if (!input) {
  console.log('Please provide a value');
}
